Evaluasi Postfix

Cara menghitung hasil dari ekspresi postfix (seperti 4 5 6 * +), Baca dari kiri ke kanan. Jika ketemu angka, simpan di tumpukan.Jika ketemu operator, ambil dua angka terakhir dari tumpukan, hitung, lalu simpan lagi hasilnya di tumpukan.Angka terakhir yang tersisa di tumpukan adalah jawabannya.

Implementasi Python:

python 
soal = input("Masukkan Infix: ")

soal_list = list(soal[::-1])
for i in range(len(soal_list)):
    if soal_list[i] == '(': soal_list[i] = ')'
    elif soal_list[i] == ')': soal_list[i] = '('
soal_balik = "".join(soal_list)

stack_pre = []
hasil_pre = []
kuat = {'+': 1, '-': 1, '*': 2, '/': 2, '^': 3, '(': 0}

for k in soal_balik:
    if k.isalnum():
        hasil_pre.append(k)
    elif k == '(':
        stack_pre.append(k)
    elif k == ')':
        while stack_pre and stack_pre[-1] != '(':
            hasil_pre.append(stack_pre.pop())
        if stack_pre: stack_pre.pop()
    else:
        while stack_pre and kuat[k] < kuat.get(stack_pre[-1], 0):
            hasil_pre.append(stack_pre.pop())
        stack_pre.append(k)

while stack_pre:
    hasil_pre.append(stack_pre.pop())

prefix = "".join(hasil_pre)[::-1]

stack_post = []
hasil_post = []

for k in soal:
    if k.isalnum():
        hasil_post.append(k)
    elif k == '(':
        stack_post.append(k)
    elif k == ')':
        while stack_post and stack_post[-1] != '(':
            hasil_post.append(stack_post.pop())
        if stack_post: stack_post.pop()
    else:

        while stack_post and kuat[k] <= kuat.get(stack_post[-1], 0):
            hasil_post.append(stack_post.pop())
        stack_post.append(k)

while stack_post:
    hasil_post.append(stack_post.pop())

postfix = "".join(hasil_post)
print("Hasil Prefix  :", prefix)
print("Hasil Postfix :", postfix)
